1. Left and Right Rotation of an Array

Question: Implement a program for left rotation and right rotation of array by a given number k.

  • Viva & Theory: Shifting one element at a time avoids extra memory (O(1) space). Always take k = k % n first; if k equals array size n, the array returns to its original state.
  • Complexity: Time: \(O(n \times k)\) | Space: \(O(1)\)
#include <iostream>
using namespace std;

// Left rotation by k
void leftRotate(int a[], int n, int k) {
    k %= n;

    while (k--) {
        int first = a[0];

        for (int i = 0; i < n - 1; i++)
            a[i] = a[i + 1];

        a[n - 1] = first;
    }
}

// Right rotation by k
void rightRotate(int a[], int n, int k) {
    k %= n;

    while (k--) {
        int last = a[n - 1];

        for (int i = n - 1; i > 0; i--)
            a[i] = a[i - 1];

        a[0] = last;
    }
}

int main() {
    int a[] = {1, 2, 3, 4, 5}, b[] = {1, 2, 3, 4, 5};
    int n = 5, k = 2;

    leftRotate(a, n, k);
    cout << "Left: ";
    for (int i = 0; i < n; i++)
        cout << a[i] << " ";

    rightRotate(b, n, k);
    cout << "\nRight: ";
    for (int i = 0; i < n; i++)
        cout << b[i] << " ";
    return 0;
}

2. Matrix Addition and Multiplication

Question: Implement a program for addition and multiplication of two matrices using concepts of 2-D arrays.

  • Viva & Theory: Addition requires both matrices to have identical dimensions (\(r_1 = r_2, c_1 = c_2\)). Multiplication requires columns of the first matrix to equal rows of the second (\(c_1 = r_2\)), producing a matrix of size \(r_1 \times c_2\).
  • Complexity: Addition: \(O(r \times c)\) | Multiplication: \(O(r_1 \times c_2 \times c_1)\) | Space: \(O(r \times c)\)
#include <iostream>
using namespace std;

int main() {
    int r = 2, c = 2;
    int A[2][2] = {{1, 2}, {3, 4}};
    int B[2][2] = {{5, 6}, {7, 8}};
    int sum[2][2], prod[2][2];

    // Addition
    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++)
            sum[i][j] = A[i][j] + B[i][j];
    }

    // Multiplication
    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++) {
            prod[i][j] = 0;
            for (int k = 0; k < c; k++)
                prod[i][j] += A[i][k] * B[k][j];
        }
    }

    cout << "Sum:\n";
    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++)
            cout << sum[i][j] << " ";
        cout << endl;
    }

    cout << "Product:\n";
    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++)
            cout << prod[i][j] << " ";
        cout << endl;
    }
    return 0;
}

3. Polynomial Addition using Linked List

Question: Implement a program for addition of two polynomials using concepts of linked list.

  • Viva & Theory: Linked lists are preferred over arrays for polynomials because sparse polynomials (e.g., \(5x^{100} + 2\)) waste massive memory in array representation. Terms are stored in descending order of exponents for a single-pass merge.
  • Complexity: Time: \(O(m + n)\) | Space: \(O(m + n)\) (where \(m, n\) are number of terms)
#include <iostream>
using namespace std;

struct Node {
    int coeff, exp;
    Node* next;
};

Node* createNode(int c, int e) {
    Node* n = new Node;
    n->coeff = c;
    n->exp = e;
    n->next = NULL;
    return n;
}

void insert(Node*& head, int c, int e) {
    Node* n = createNode(c, e);
    if (!head) {
        head = n;
        return;
    }
    Node* temp = head;
    while (temp->next)
        temp = temp->next;
    temp->next = n;
}

Node* addPoly(Node* p1, Node* p2) {
    Node* res = NULL;
    while (p1 && p2) {
        if (p1->exp > p2->exp) {
            insert(res, p1->coeff, p1->exp);
            p1 = p1->next;
        } else if (p1->exp < p2->exp) {
            insert(res, p2->coeff, p2->exp);
            p2 = p2->next;
        } else {
            insert(res, p1->coeff + p2->coeff, p1->exp);
            p1 = p1->next;
            p2 = p2->next;
        }
    }
    while (p1) {
        insert(res, p1->coeff, p1->exp);
        p1 = p1->next;
    }
    while (p2) {
        insert(res, p2->coeff, p2->exp);
        p2 = p2->next;
    }
    return res;
}

int main() {
    Node *p1 = NULL, *p2 = NULL;
    insert(p1, 5, 2); insert(p1, 4, 1); insert(p1, 2, 0);
    insert(p2, 3, 2); insert(p2, 1, 0);

    Node* sum = addPoly(p1, p2);
    while (sum) {
        cout << sum->coeff << "x^" << sum->exp;
        if (sum->next) cout << " + ";
        sum = sum->next;
    }
    return 0;
}

4. Merge Doubly Linked Lists at Alternate Positions & Sort

Question: Implement a program for merging a doubly linked list into another doubly linked list at alternate positions and sorting the resultant doubly linked list.

  • Viva & Theory: In a Doubly Linked List (DLL), every pointer rewiring requires updating both next and prev links. After alternating insertion, if the second list has leftover nodes, they are appended at the tail before sorting.
  • Complexity: Merge: \(O(\min(m, n))\) | Bubble Sort: \(O((m + n)^2)\) | Space: \(O(1)\) (in-place)
#include <iostream>
using namespace std;

struct Node {
    int data;
    Node *prev, *next;
};

void insert(Node*& head, int val) {
    Node* n = new Node{val, NULL, NULL};
    if (!head) {
        head = n;
        return;
    }
    Node* temp = head;
    while (temp->next)
        temp = temp->next;
    temp->next = n;
    n->prev = temp;
}

void mergeAlt(Node*& h1, Node*& h2) {
    Node *c1 = h1, *c2 = h2;
    while (c1 && c2) {
        Node* n1 = c1->next;
        Node* n2 = c2->next;

        c2->next = n1;
        if (n1) n1->prev = c2;
        c1->next = c2;
        c2->prev = c1;

        c1 = n1;
        c2 = n2;
    }
    // Append remaining nodes if h2 was longer
    if (c2) {
        Node* tail = h1;
        while (tail->next) tail = tail->next;
        tail->next = c2;
        c2->prev = tail;
    }
}

void sortDLL(Node* head) {
    if (!head) return;
    bool swapped;
    do {
        swapped = false;
        Node* curr = head;
        while (curr->next) {
            if (curr->data > curr->next->data) {
                int t = curr->data;
                curr->data = curr->next->data;
                curr->next->data = t;
                swapped = true;
            }
            curr = curr->next;
        }
    } while (swapped);
}

int main() {
    Node *l1 = NULL, *l2 = NULL;
    insert(l1, 30); insert(l1, 10);
    insert(l2, 40); insert(l2, 20); insert(l2, 5);

    mergeAlt(l1, l2);
    sortDLL(l1);

    for (Node* t = l1; t; t = t->next)
        cout << t->data << " ";
    return 0;
}

5. Railway Reservation using Circular Linked List

Question: Implement railway reservation system using a circular linked list.

  • Viva & Theory: In a Circular Linked List (CLL), the next pointer of the last node points back to head instead of NULL. Traversal always uses a do-while loop terminating when curr == head to prevent infinite loops.
  • Complexity: Book / Cancel / Display: \(O(n)\) | Space: \(O(n)\)
#include <iostream>
#include <string>
using namespace std;

struct Seat {
    int no;
    string name;
    bool booked;
    Seat* next;
};

Seat* createTrain(int total) {
    Seat *head = NULL, *prev = NULL;
    for (int i = 1; i <= total; i++) {
        Seat* s = new Seat{i, "Empty", false, NULL};
        if (!head) head = s;
        else prev->next = s;
        prev = s;
    }
    prev->next = head; // Circular link
    return head;
}

void bookSeat(Seat* head, int no, string pname) {
    Seat* curr = head;
    do {
        if (curr->no == no) {
            if (curr->booked) cout << "Already booked!\n";
            else {
                curr->booked = true;
                curr->name = pname;
            }
            return;
        }
        curr = curr->next;
    } while (curr != head);
}

void cancelSeat(Seat* head, int no) {
    Seat* curr = head;
    do {
        if (curr->no == no) {
            curr->booked = false;
            curr->name = "Empty";
            return;
        }
        curr = curr->next;
    } while (curr != head);
}

void display(Seat* head) {
    Seat* curr = head;
    do {
        cout << "Seat " << curr->no << ": " << curr->name << endl;
        curr = curr->next;
    } while (curr != head);
}

int main() {
    Seat* train = createTrain(3);
    bookSeat(train, 1, "Rahul");
    bookSeat(train, 2, "Priya");
    cancelSeat(train, 1);
    display(train);
    return 0;
}

6. 8 Queens Problem using Stack

Question: Implement 8 Queens problem using stack.

  • Viva & Theory: Two queens attack each other if they share the same row, column, or diagonal (\(\vert{}r_1 - r_2\vert{} == \vert{}c_1 - c_2\vert{}\)). Instead of recursion (which uses the system call stack), an explicit array-based stack stores the chosen column for each row to enable manual backtracking.
  • Complexity: Time: \(O(N!)\) | Space: \(O(N)\)
#include <iostream>
using namespace std;

#define N 8

int st[N], top = -1; // Manual stack storing column index for each row

void push(int col) { st[++top] = col; }
int pop() { return st[top--]; }

bool isSafe(int row, int col) {
    for (int r = 0; r < row; r++) {
        int c = st[r];
        int rowDiff = row - r;
        int colDiff = (col > c) ? (col - c) : (c - col);
        if (c == col || rowDiff == colDiff)
            return false;
    }
    return true;
}

int main() {
    int row = 0, col = 0;

    while (row < N) {
        while (col < N && !isSafe(row, col))
            col++;

        if (col < N) {
            push(col);
            row++;
            col = 0;
        } else {
            if (top == -1) break;
            row--;
            col = pop() + 1; // Backtrack to next column
        }
    }

    for (int r = 0; r < N; r++) {
        for (int c = 0; c < N; c++)
            cout << (st[r] == c ? "Q " : ". ");
        cout << endl;
    }
    return 0;
}

7. Tower of Hanoi using Recursion

Question: Implement Tower of Hanoi problem using concept of recursion.

  • Viva & Theory: Classic divide-and-conquer recursion. To move \(n\) disks from Source (A) to Destination (C) using Auxiliary (B): move \(n-1\) disks A -> B, move the largest disk A -> C, then move \(n-1\) disks B -> C. Total moves required = \(2^n - 1\).
  • Complexity: Time: \(O(2^n)\) | Space: \(O(n)\) (recursion stack depth)
#include <iostream>
using namespace std;

void toh(int n, char src, char aux, char dest) {
    if (n == 1) {
        cout << "Move disk 1 from " << src << " to " << dest << endl;
        return;
    }
    toh(n - 1, src, dest, aux);
    cout << "Move disk " << n << " from " << src << " to " << dest << endl;
    toh(n - 1, aux, src, dest);
}

int main() {
    int n = 3;
    toh(n, 'A', 'B', 'C');
    return 0;
}

8. Hospital Priority Queue

Question: Implement basic operations of priority queue for managing the appointments of patients at a hospital.

  • Viva & Theory: Unlike a standard FIFO queue, a Priority Queue dequeues elements based on priority rank rather than arrival order. Keeping the array sorted during insertion makes dequeue \(O(1)\) and enqueue \(O(n)\).
  • Complexity: Enqueue: \(O(n)\) | Dequeue: \(O(1)\) | Space: \(O(n)\)
#include <iostream>
#include <string>
using namespace std;

struct Patient {
    string name;
    int priority; // Higher number = higher urgency
};

Patient pq[10];
int n = 0;

void enqueue(string name, int prio) {
    int i = n - 1;
    // Shift lower-priority patients right
    while (i >= 0 && pq[i].priority > prio) {
        pq[i + 1] = pq[i];
        i--;
    }
    pq[i + 1] = {name, prio};
    n++;
}

void dequeue() {
    if (n == 0) {
        cout << "No patients waiting.\n";
        return;
    }
    // Highest priority is at the end (n - 1)
    cout << "Treating: " << pq[n - 1].name << " (Priority " << pq[n - 1].priority << ")\n";
    n--;
}

int main() {
    enqueue("Amit (Fever)", 1);
    enqueue("Sneha (Emergency)", 3);
    enqueue("Raj (Fracture)", 2);

    dequeue();
    dequeue();
    dequeue();
    return 0;
}

9. Non-Preemptive FCFS and SJF Scheduling using Queue

Question: Implement FCFS and SJF scheduling in non-preemptive mode using suitable queue.

  • Viva & Theory: FCFS uses a standard FIFO queue and suffers from the Convoy Effect (short jobs wait behind a long CPU-bound job). SJF minimizes average waiting time by executing the shortest burst time first.
  • Complexity: FCFS: \(O(n)\) | SJF: \(O(n^2)\) (due to sorting) | Space: \(O(n)\)
#include <iostream>
using namespace std;

struct Process {
    int id, bt;
};

// Simple circular/linear queue from scratch
struct Queue {
    Process arr[10];
    int front = 0, rear = -1;
    void push(Process p) { arr[++rear] = p; }
    Process pop() { return arr[front++]; }
    bool empty() { return front > rear; }
};

void simulate(Queue q, string title) {
    cout << "\n--- " << title << " ---\n";
    int wt = 0, totalWt = 0, count = 0;
    while (!q.empty()) {
        Process p = q.pop();
        cout << "P" << p.id << " | Burst: " << p.bt << " | Wait: " << wt << endl;
        totalWt += wt;
        wt += p.bt;
        count++;
    }
    cout << "Avg Wait Time: " << (float)totalWt / count << endl;
}

int main() {
    Process p[] = {{1, 6}, {2, 2}, {3, 8}, {4, 3}};
    int n = 4;

    // FCFS
    Queue fcfs;
    for (int i = 0; i < n; i++)
        fcfs.push(p[i]);
    simulate(fcfs, "FCFS");

    // SJF: Sort by burst time first
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (p[j].bt > p[j + 1].bt) {
                Process t = p[j];
                p[j] = p[j + 1];
                p[j + 1] = t;
            }
        }
    }

    Queue sjf;
    for (int i = 0; i < n; i++)
        sjf.push(p[i]);
    simulate(sjf, "SJF");
    return 0;
}

10. Palindrome Check using Deque

Question: Implement a program to check whether the given string is palindrome or not using doubly ended queue.

  • Viva & Theory: A Deque (Double-Ended Queue) allows insertion and deletion from both front and rear. Popping characters simultaneously from both ends and comparing them verifies a palindrome in \(\lfloor n/2 \rfloor\) comparisons.
  • Complexity: Time: \(O(n)\) | Space: \(O(n)\)
#include <iostream>
#include <string>
using namespace std;

class Deque {
    char arr[50];
    int front = 0, rear = -1;
public:
    void insertRear(char ch) { arr[++rear] = ch; }
    char deleteFront() { return arr[front++]; }
    char deleteRear() { return arr[rear--]; }
    int size() { return rear - front + 1; }
};

bool isPalindrome(string s) {
    Deque dq;
    for (int i = 0; i < s.length(); i++)
        dq.insertRear(s[i]);

    while (dq.size() > 1) {
        if (dq.deleteFront() != dq.deleteRear())
            return false;
    }
    return true;
}

int main() {
    string str = "madam";
    if (isPalindrome(str))
        cout << str << " is a Palindrome\n";
    else
        cout << str << " is NOT a Palindrome\n";
    return 0;
}

11. Phone Book with Sentinel Search & Sorted Insertion

Question: Implement a program to store names and mobile numbers of your friends in sorted order on names. Search your friend from list using Sentinel search. Insert friend if not present in phone book.

  • Viva & Theory: Sentinel Search improves linear search by placing the search key at the last index (arr[n-1]). This guarantees the element will be found, eliminating the i < n boundary check inside every iteration of the while loop.
  • Complexity: Sentinel Search: \(O(n)\) | Sorted Insertion: \(O(n)\) | Space: \(O(1)\)
#include <iostream>
#include <string>
using namespace std;

struct Friend {
    string name, mobile;
};

void insertSorted(Friend pb[], int& n, string name, string mob) {
    int i = n - 1;
    while (i >= 0 && pb[i].name > name) {
        pb[i + 1] = pb[i];
        i--;
    }
    pb[i + 1] = {name, mob};
    n++;
}

int sentinelSearch(Friend pb[], int n, string key) {
    if (n == 0) return -1;
    Friend last = pb[n - 1];
    pb[n - 1].name = key; // Set sentinel

    int i = 0;
    while (pb[i].name != key) // Only 1 comparison per iteration
        i++;

    pb[n - 1] = last; // Restore original last element
    if (i < n - 1 || pb[n - 1].name == key)
        return i;
    return -1;
}

int main() {
    Friend pb[10];
    int n = 0;

    insertSorted(pb, n, "Rohan", "9876543210");
    insertSorted(pb, n, "Amit", "9123456780");

    string searchName = "Neha";
    int idx = sentinelSearch(pb, n, searchName);

    if (idx != -1)
        cout << "Found: " << pb[idx].mobile << endl;
    else {
        cout << searchName << " not found. Adding...\n";
        insertSorted(pb, n, searchName, "9988776655");
    }

    for (int i = 0; i < n; i++)
        cout << pb[i].name << " : " << pb[i].mobile << endl;
    return 0;
}

12. Student Percentages using Merge Sort

Question: Implement a program to store percentage of second year students in array. Write a function for sorting array of floating point numbers in ascending order using Merge Sort.

  • Viva & Theory: Merge Sort is a stable, divide-and-conquer sorting algorithm. Unlike Quick Sort (which degrades to \(O(n^2)\) on sorted inputs), Merge Sort guarantees \(O(n \log n)\) in all cases at the cost of \(O(n)\) auxiliary memory.
  • Complexity: Time: \(O(n \log n)\) (Best, Average, Worst) | Space: \(O(n)\)
#include <iostream>
using namespace std;

void merge(float a[], int low, int mid, int high) {
    float temp[50];
    int i = low, j = mid + 1, k = low;

    while (i <= mid && j <= high) {
        if (a[i] <= a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }
    while (i <= mid) temp[k++] = a[i++];
    while (j <= high) temp[k++] = a[j++];

    for (int x = low; x <= high; x++)
        a[x] = temp[x];
}

void mergeSort(float a[], int low, int high) {
    if (low >= high) return;
    int mid = (low + high) / 2;
    mergeSort(a, low, mid);
    mergeSort(a, mid + 1, high);
    merge(a, low, mid, high);
}

int main() {
    float perc[] = {78.5, 91.2, 65.0, 88.4, 72.1};
    int n = 5;

    mergeSort(perc, 0, n - 1);

    for (int i = 0; i < n; i++)
        cout << perc[i] << "% ";
    return 0;
}

13. Separate Chaining with Replacement

Question: Scores of batsman in twelve overs need to be stored in a table using hash function: hash(key) = key mod 10. Write a C++ program to implement hash table for collision handling using separate chaining with replacement using linked list for the following data: 10, 5, 4, 14, 15, 20, 1, 11, 6, 16, 36, 28.

  • Viva & Theory: In with replacement, every bucket’s head node is reserved for a key whose true home address is that bucket. If a bucket is occupied by a displaced node (home != idx), the new rightful key evicts it to its true home chain so primary lookups remain \(O(1)\).
  • Complexity: Average Search/Insert: \(O(1 + \alpha)\) | Worst Case: \(O(n)\) | Space: \(O(n)\)
#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* next;
};

Node* table[10];

int hashFunc(int key) { return key % 10; }

void appendNode(int idx, Node* node) {
    node->next = NULL;
    if (!table[idx]) {
        table[idx] = node;
        return;
    }
    Node* temp = table[idx];
    while (temp->next)
        temp = temp->next;
    temp->next = node;
}

void insert(int key) {
    int idx = hashFunc(key);
    Node* n = new Node{key, NULL};

    if (!table[idx]) {
        table[idx] = n;
        return;
    }

    int existingHome = hashFunc(table[idx]->data);
    if (existingHome != idx) {
        // Displace misplaced head node to its rightful bucket
        Node* displaced = table[idx];
        n->next = displaced->next;
        table[idx] = n;
        appendNode(existingHome, displaced);
    } else {
        appendNode(idx, n);
    }
}

int main() {
    for (int i = 0; i < 10; i++) table[i] = NULL;
    int data[] = {10, 5, 4, 14, 15, 20, 1, 11, 6, 16, 36, 28};

    for (int i = 0; i < 12; i++)
        insert(data[i]);

    for (int i = 0; i < 10; i++) {
        cout << i << ": ";
        for (Node* t = table[i]; t; t = t->next)
            cout << t->data << " -> ";
        cout << "NULL\n";
    }
    return 0;
}

14. Linear Probing without Replacement

Question: Vinay has a medical shop and wants to store information of medical stock available in a table using hash function: hash(key) = key mod 9. Write a C++ program to implement hash table for collision handling using linear probing without replacement.

  • Viva & Theory: In linear probing without replacement, if the home index is occupied (even by a colliding element from another index), we probe sequentially using (idx + 1) % 9 until an empty slot (-1) is found. Its main drawback is primary clustering (long contiguous blocks of occupied slots).
  • Complexity: Average Search/Insert: \(O(1)\) | Worst Case: \(O(n)\) | Space: \(O(m)\) (table size)
#include <iostream>
using namespace std;

#define SIZE 9
int table[SIZE];

int hashFunc(int key) { return key % SIZE; }

void insert(int key) {
    int idx = hashFunc(key);
    int start = idx;

    while (table[idx] != -1) {
        idx = (idx + 1) % SIZE;
        if (idx == start) {
            cout << "Table Full!\n";
            return;
        }
    }
    table[idx] = key;
}

int main() {
    for (int i = 0; i < SIZE; i++)
        table[i] = -1;

    int stock[] = {18, 27, 10, 19, 36, 45, 5};
    for (int i = 0; i < 7; i++)
        insert(stock[i]);

    for (int i = 0; i < SIZE; i++)
