1. BCD to HEX

Pure assembly (_start, syscalls). Link with ld.

; ============================================================
; Program   : BCD to Hexadecimal Conversion
; Platform  : Linux x86-64
; Assembler : NASM
;
; Converts an input BCD (decimal) string into a 64-bit integer
; and displays its 16-digit hexadecimal representation.
; ============================================================



section .data
    ; User prompts and output labels
    msg1        db "Enter a BCD Number: ", 10
    len1        equ $ - msg1

    msg2        db "HEX Equivalent is : ", 10
    len2        equ $ - msg2

    newline     db 10



section .bss
    ; Input buffer and calculation variables
    char_buff   resb 17
    cnt         resq 1
    result      resq 1



section .text
    global _start


; ============================================================
; MACRO: READ
; Purpose: Read characters from standard input.
;
; Parameters:
;   %1 = Address of receiving buffer
;   %2 = Maximum byte count to read
; ============================================================
%macro READ 2
    mov rax, 0
    mov rdi, 0
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro


; ============================================================
; MACRO: WRITE
; Purpose: Display a string to standard output.
;
; Parameters:
;   %1 = Address of string buffer
;   %2 = Length of string in bytes
; ============================================================
%macro WRITE 2
    mov rax, 1
    mov rdi, 1
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro


_start:
    ; --------------------------------------------------------
    ; 1. Read BCD input string
    ; --------------------------------------------------------
    WRITE msg1, len1
    READ char_buff, 17

    ; Exclude trailing newline from character count
    dec rax
    mov [cnt], rax

    mov rsi, char_buff
    mov rcx, [cnt]
    xor rbx, rbx


    ; --------------------------------------------------------
    ; 2. Convert ASCII BCD digits into numeric value
    ; --------------------------------------------------------
BCD_CONVERT_LOOP:
    mov rax, 10
    mul rbx
    mov rbx, rax

    xor rdx, rdx
    mov dl, byte [rsi]
    sub dl, '0'
    add rbx, rdx

    inc rsi
    dec rcx
    jnz BCD_CONVERT_LOOP

    mov [result], rbx


    ; --------------------------------------------------------
    ; 3. Display hexadecimal result and exit
    ; --------------------------------------------------------
    WRITE msg2, len2

    mov rbx, [result]
    call DISPLAY_HEX

    WRITE newline, 1
    jmp EXIT_PROGRAM


; ============================================================
; PROCEDURE: DISPLAY_HEX
; Purpose: Prints the 64-bit value in RBX as 16 hexadecimal digits.
;
; Input : RBX = number to display
; ============================================================
DISPLAY_HEX:
    mov rsi, char_buff
    mov rcx, 16


DISPLAY_HEX_LOOP:
    rol rbx, 4

    mov dl, bl
    and dl, 0x0F

    cmp dl, 9
    jbe FORMAT_DEC_DIGIT

    add dl, 'A' - 10
    jmp STORE_HEX_DIGIT


FORMAT_DEC_DIGIT:
    add dl, '0'


STORE_HEX_DIGIT:
    mov byte [rsi], dl
    inc rsi
    dec rcx
    jnz DISPLAY_HEX_LOOP

    WRITE char_buff, 16
    ret


; ============================================================
; EXIT_PROGRAM
; Purpose: Terminate the program cleanly.
; ============================================================
EXIT_PROGRAM:
    mov rax, 60
    xor rdi, rdi
    syscall



section .note.GNU-stack noalloc noexec nowrite progbits
nasm -f elf64 bcd2hex.asm
ld -o bcd2hex bcd2hex.o
./bcd2hex

2. HEX to BCD

Pure assembly (_start, syscalls). Link with ld.

; ============================================================
; Program   : Hexadecimal to BCD Conversion
; Platform  : Linux x86-64
; Assembler : NASM
;
; Converts a 64-bit hexadecimal input string into its numeric
; value and prints its decimal (BCD) digit sequence.
; ============================================================



section .data
    ; User prompts and output labels
    msg1        db "Enter the HEX number: ", 10
    len1        equ $ - msg1

    msg2        db "The BCD Equivalent : ", 10
    len2        equ $ - msg2

    newline     db 10



section .bss
    ; Buffers and conversion counters
    char_buff   resb 17
    len         resq 1
    char        resb 1
    cnt         resb 1



section .text
    global _start


; ============================================================
; MACRO: READ
; Purpose: Read characters from standard input.
;
; Parameters:
;   %1 = Address of receiving buffer
;   %2 = Maximum byte count to read
; ============================================================
%macro READ 2
    mov rax, 0
    mov rdi, 0
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro


; ============================================================
; MACRO: WRITE
; Purpose: Display a string to standard output.
;
; Parameters:
;   %1 = Address of string buffer
;   %2 = Length of string in bytes
; ============================================================
%macro WRITE 2
    mov rax, 1
    mov rdi, 1
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro


_start:
    ; --------------------------------------------------------
    ; 1. Read hexadecimal input and convert to binary
    ; --------------------------------------------------------
    WRITE msg1, len1
    READ char_buff, 17

    dec rax
    mov [len], rax

    mov rcx, [len]
    call ACCEPT_HEX


    ; --------------------------------------------------------
    ; 2. Extract decimal digits onto the stack
    ; --------------------------------------------------------
    xor rcx, rcx
    mov rax, rbx
    mov rbx, 10


DIVIDE_DECIMAL_LOOP:
    xor rdx, rdx
    div rbx

    push rdx
    inc rcx

    test rax, rax
    jnz DIVIDE_DECIMAL_LOOP

    mov byte [cnt], cl


    ; --------------------------------------------------------
    ; 3. Pop and print each decimal digit
    ; --------------------------------------------------------
    WRITE msg2, len2


PRINT_DECIMAL_LOOP:
    pop rbx
    add bl, '0'
    mov byte [char], bl

    WRITE char, 1

    dec byte [cnt]
    jnz PRINT_DECIMAL_LOOP

    WRITE newline, 1
    jmp EXIT_PROGRAM


; ============================================================
; PROCEDURE: ACCEPT_HEX
; Purpose: Converts an ASCII hexadecimal string from char_buff
;          into an unsigned 64-bit value in RBX.
;
; Input : char_buff, RCX = character count
; Output: RBX = numeric value
; ============================================================
ACCEPT_HEX:
    mov rsi, char_buff
    xor rbx, rbx


ACCEPT_HEX_LOOP:
    xor rdx, rdx
    mov dl, byte [rsi]

    cmp dl, '9'
    jbe SUBTRACT_ZERO

    sub dl, 7


SUBTRACT_ZERO:
    sub dl, '0'
    shl rbx, 4
    add rbx, rdx

    inc rsi
    dec rcx
    jnz ACCEPT_HEX_LOOP

    ret


; ============================================================
; EXIT_PROGRAM
; Purpose: Terminate the program cleanly.
; ============================================================
EXIT_PROGRAM:
    mov rax, 60
    xor rdi, rdi
    syscall



section .note.GNU-stack noalloc noexec nowrite progbits
nasm -f elf64 hex2bcd.asm
ld -o hex2bcd hex2bcd.o
./hex2bcd

3. Accept 64-bit Numbers & Perform Arithmetic Operations

Pure assembly. Link with ld.

; ============================================================
; Program   : 64-bit Hexadecimal Arithmetic Operations
; Platform  : Linux x86-64
; Assembler : NASM
;
; Options:
;   1. Addition
;   2. Subtraction
;   3. Multiplication
;   4. Division (Quotient and Remainder)
;   5. Exit
; ============================================================



section .data
    ; Menu text and option prompts
    menu        db 10, "--- MENU ---", 10
                db "1. Addition", 10
                db "2. Subtraction", 10
                db "3. Multiplication", 10
                db "4. Division", 10
                db "5. Exit", 10
                db "Enter your choice: "
    menulen     equ $ - menu

    ; Input prompts and result messages
    msg1        db "Enter 1st 64-bit Hex number: "
    len1        equ $ - msg1

    msg2        db "Enter 2nd 64-bit Hex number: "
    len2        equ $ - msg2

    msg3        db "Result is: "
    len3        equ $ - msg3

    msg4        db "Remainder is: "
    len4        equ $ - msg4

    newline     db 10



section .bss
    ; Buffers and 64-bit operands/results
    char_buff   resb 17
    choice      resb 2
    a           resq 1
    b           resq 1
    c           resq 1
    rem         resq 1



section .text
    global _start


; ============================================================
; MACRO: READ
; Purpose: Read characters from standard input.
;
; Parameters:
;   %1 = Address of receiving buffer
;   %2 = Maximum byte count to read
; ============================================================
%macro READ 2
    mov rax, 0
    mov rdi, 0
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro


; ============================================================
; MACRO: WRITE
; Purpose: Display a string to standard output.
;
; Parameters:
;   %1 = Address of string buffer
;   %2 = Length of string in bytes
; ============================================================
%macro WRITE 2
    mov rax, 1
    mov rdi, 1
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro


_start:
    ; --------------------------------------------------------
    ; Display menu and route user selection
    ; --------------------------------------------------------
UP_MENU:
    WRITE menu, menulen
    READ choice, 2

    cmp byte [choice], '5'
    je EXIT_PROGRAM

    cmp byte [choice], '1'
    je ADDITION_OP

    cmp byte [choice], '2'
    je SUBTRACTION_OP

    cmp byte [choice], '3'
    je MULTIPLICATION_OP

    cmp byte [choice], '4'
    je DIVISION_OP

    jmp UP_MENU


; ============================================================
; OPTION 1: ADDITION
; ============================================================
ADDITION_OP:
    call ACCEPT_NUMS

    mov rax, [a]
    add rax, [b]
    mov [c], rax

    WRITE msg3, len3
    mov rbx, [c]
    call DISPLAY_HEX
    WRITE newline, 1

    jmp UP_MENU


; ============================================================
; OPTION 2: SUBTRACTION
; ============================================================
SUBTRACTION_OP:
    call ACCEPT_NUMS

    mov rax, [a]
    sub rax, [b]
    mov [c], rax

    WRITE msg3, len3
    mov rbx, [c]
    call DISPLAY_HEX
    WRITE newline, 1

    jmp UP_MENU


; ============================================================
; OPTION 3: MULTIPLICATION
; ============================================================
MULTIPLICATION_OP:
    call ACCEPT_NUMS

    mov rax, [a]
    mov rbx, [b]
    mul rbx
    mov [c], rax

    WRITE msg3, len3
    mov rbx, [c]
    call DISPLAY_HEX
    WRITE newline, 1

    jmp UP_MENU


; ============================================================
; OPTION 4: DIVISION
; ============================================================
DIVISION_OP:
    call ACCEPT_NUMS

    xor rdx, rdx
    mov rax, [a]
    mov rbx, [b]
    div rbx

    mov [c], rax
    mov [rem], rdx

    WRITE msg3, len3
    mov rbx, [c]
    call DISPLAY_HEX
    WRITE newline, 1

    WRITE msg4, len4
    mov rbx, [rem]
    call DISPLAY_HEX
    WRITE newline, 1

    jmp UP_MENU


; ============================================================
; PROCEDURE: ACCEPT_NUMS
; Purpose: Prompts for and reads two 64-bit hexadecimal operands.
; ============================================================
ACCEPT_NUMS:
    WRITE msg1, len1
    READ char_buff, 17
    dec rax
    mov rcx, rax
    call ACCEPT_HEX
    mov [a], rbx

    WRITE msg2, len2
    READ char_buff, 17
    dec rax
    mov rcx, rax
    call ACCEPT_HEX
    mov [b], rbx

    ret


; ============================================================
; PROCEDURE: ACCEPT_HEX
; Purpose: Converts an ASCII hexadecimal string into a 64-bit value.
;
; Input : char_buff, RCX = character count
; Output: RBX = 64-bit unsigned integer
; ============================================================
ACCEPT_HEX:
    mov rsi, char_buff
    xor rbx, rbx


ACCEPT_HEX_LOOP:
    xor rdx, rdx
    mov dl, byte [rsi]

    cmp dl, '9'
    jbe SUBTRACT_ZERO

    sub dl, 7


SUBTRACT_ZERO:
    sub dl, '0'
    shl rbx, 4
    add rbx, rdx

    inc rsi
    dec rcx
    jnz ACCEPT_HEX_LOOP

    ret


; ============================================================
; PROCEDURE: DISPLAY_HEX
; Purpose: Displays the 64-bit value in RBX as 16 hexadecimal digits.
;
; Input : RBX = number to display
; ============================================================
DISPLAY_HEX:
    mov rsi, char_buff
    mov rcx, 16


DISPLAY_HEX_LOOP:
    rol rbx, 4

    mov dl, bl
    and dl, 0x0F

    cmp dl, 9
    jbe FORMAT_DEC_DIGIT

    add dl, 'A' - 10
    jmp STORE_HEX_DIGIT

    inc rsi
    dec rcx
    jnz DISPLAY_HEX_LOOP

    WRITE char_buff, 16
    ret


; ============================================================
; EXIT_PROGRAM

; ============================================================

4. String Operations

Pure assembly. Link with ld.

; ============================================================
; Program   : String Operations
; Platform  : Linux x86-64
; Assembler : NASM
;
; Options:
;   1. String length
;   2. String copy
;   3. String reverse
;   4. String compare
;   5. String concatenate
;   6. String palindrome
;   7. Substring search
;   8. Exit
; ============================================================



section .data
    ; Menu options and choice prompt
    menu        db 10, " ", 10
                db "1..String length", 10
                db "2..String copy", 10
                db "3..String reverse", 10
                db "4..String compare", 10
                db "5..String concatenate", 10
                db "6..String palindrome", 10
                db "7..Substring", 10
                db "8..Exit", 10
    menu_len    equ $ - menu

    ch_msg      db "Enter your choice : ", 10
    ch_len      equ $ - ch_msg

    ; User input prompts and status outputs
    msg1        db "Enter String 1 : ", 10
    len1        equ $ - msg1

    msg2        db "Enter String 2 : ", 10
    len2        equ $ - msg2

    msg3        db "Answer : ", 10
    len3        equ $ - msg3

    eq_msg      db "Strings are Equal", 10
    eq_len      equ $ - eq_msg

    neq_msg     db "Strings are NOT Equal", 10
    neq_len     equ $ - neq_msg

    pal_msg     db "String is Palindrome", 10
    pal_len     equ $ - pal_msg

    npal_msg    db "String is NOT Palindrome", 10
    npal_len    equ $ - npal_msg

    sub_msg     db "Substring Found", 10
    sub_len     equ $ - sub_msg

    nsub_msg    db "Substring NOT Found", 10
    nsub_len    equ $ - nsub_msg

    newline     db 10



section .bss
    ; String buffers and lengths
    char_buff   resb 17
    string1     resb 20
    string2     resb 20
    string3     resb 40
    choice      resb 2

    l1          resq 1
    l2          resq 1
    l3          resq 1



section .text
    global _start


; ============================================================
; MACRO: READ
; Purpose: Read characters from standard input.
;
; Parameters:
;   %1 = Address of receiving buffer
;   %2 = Maximum byte count to read
; ============================================================
%macro READ 2
    mov rax, 0
    mov rdi, 0
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro


; ============================================================
; MACRO: WRITE
; Purpose: Display a string to standard output.
;
; Parameters:
;   %1 = Address of string buffer
;   %2 = Length of string in bytes
; ============================================================
%macro WRITE 2
    mov rax, 1
    mov rdi, 1
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro


_start:
    ; --------------------------------------------------------
    ; Read primary input string
    ; --------------------------------------------------------
    WRITE msg1, len1
    READ string1, 20

    dec rax
    mov [l1], rax


PRINT_MENU:
    ; --------------------------------------------------------
    ; Display menu and dispatch selection
    ; --------------------------------------------------------
    WRITE menu, menu_len
    WRITE ch_msg, ch_len
    READ choice, 2

    cmp byte [choice], '1'
    je STRLEN_OP

    cmp byte [choice], '2'
    je STRCPY_OP

    cmp byte [choice], '3'
    je STRREV_OP

    cmp byte [choice], '4'
    je STRCMP_OP

    cmp byte [choice], '5'
    je STRCAT_OP

    cmp byte [choice], '6'
    je STRPAL_OP

    cmp byte [choice], '7'
    je SUBSTR_OP

    cmp byte [choice], '8'
    je EXIT_PROGRAM

    jmp PRINT_MENU


; ============================================================
; OPTION 1: STRING LENGTH
; ============================================================
STRLEN_OP:
    WRITE msg3, len3

    mov rbx, [l1]
    call DISPLAY_HEX

    WRITE newline, 1
    jmp PRINT_MENU


; ============================================================
; OPTION 2: STRING COPY
; ============================================================
STRCPY_OP:
    mov rsi, string1
    mov rdi, string3
    mov rcx, [l1]

    cld
    rep movsb

    WRITE msg3, len3
    WRITE string3, qword [l1]
    WRITE newline, 1

    jmp PRINT_MENU


; ============================================================
; OPTION 3: STRING REVERSE
; ============================================================
STRREV_OP:
    ; Point RSI to last character of string1
    mov rsi, string1
    add rsi, [l1]
    dec rsi

    mov rdi, string3
    mov rcx, [l1]


REV_LOOP:
    mov al, byte [rsi]
    mov byte [rdi], al

    dec rsi
    inc rdi
    dec rcx
    jnz REV_LOOP

    WRITE msg3, len3
    WRITE string3, qword [l1]
    WRITE newline, 1

    jmp PRINT_MENU


; ============================================================
; OPTION 4: STRING COMPARE
; ============================================================
STRCMP_OP:
    WRITE msg2, len2
    READ string2, 20

    dec rax
    mov [l2], rax

    mov rax, [l1]
    cmp rax, [l2]
    jne STRINGS_NOT_EQUAL

    mov rsi, string1
    mov rdi, string2
    mov rcx, [l1]

    cld
    repe cmpsb
    jne STRINGS_NOT_EQUAL

    WRITE eq_msg, eq_len
    jmp PRINT_MENU


STRINGS_NOT_EQUAL:
    WRITE neq_msg, neq_len
    jmp PRINT_MENU


; ============================================================
; OPTION 5: STRING CONCATENATE
; ============================================================
STRCAT_OP:
    WRITE msg2, len2
    READ string2, 20

    dec rax
    mov [l2], rax

    ; Copy string1 into string3
    mov rsi, string1
    mov rdi, string3
    mov rcx, [l1]

    cld
    rep movsb

    ; Append string2 right after string1 in string3
    mov rsi, string2
    mov rcx, [l2]

    cld
    rep movsb

    ; Total length = l1 + l2
    mov rax, [l1]
    add rax, [l2]
    mov [l3], rax

    WRITE msg3, len3
    WRITE string3, qword [l3]
    WRITE newline, 1

    jmp PRINT_MENU


; ============================================================
; OPTION 6: STRING PALINDROME (Reverse + Compare)
; ============================================================
STRPAL_OP:
    ; Reverse string1 into string3
    mov rsi, string1
    add rsi, [l1]
    dec rsi

    mov rdi, string3
    mov rcx, [l1]


PAL_REV_LOOP:
    mov al, byte [rsi]
    mov byte [rdi], al

    dec rsi
    inc rdi
    dec rcx
    jnz PAL_REV_LOOP

    ; Compare string1 with reversed string3
    mov rsi, string1
    mov rdi, string3
    mov rcx, [l1]

    cld
    repe cmpsb
    jne NOT_PALINDROME

    WRITE pal_msg, pal_len
    jmp PRINT_MENU


NOT_PALINDROME:
    WRITE npal_msg, npal_len
    jmp PRINT_MENU


; ============================================================
; OPTION 7: SUBSTRING SEARCH
; ============================================================
SUBSTR_OP:
    WRITE msg2, len2
    READ string2, 20

    dec rax
    mov [l2], rax

    ; Check if string2 is longer than string1
    mov rax, [l1]
    sub rax, [l2]
    jb SUBSTRING_NOT_FOUND

    ; Number of starting positions to check = l1 - l2 + 1
    inc rax
    mov rdx, rax
    mov rbx, string1


SUBSTR_OUTER_LOOP:
    mov rsi, rbx
    mov rdi, string2
    mov rcx, [l2]

    cld
    repe cmpsb
    je SUBSTRING_FOUND

    inc rbx
    dec rdx
    jnz SUBSTR_OUTER_LOOP


SUBSTRING_NOT_FOUND:
    WRITE nsub_msg, nsub_len
    jmp PRINT_MENU


SUBSTRING_FOUND:
    WRITE sub_msg, sub_len
    jmp PRINT_MENU


; ============================================================
; PROCEDURE: DISPLAY_HEX
; Purpose: Displays the 64-bit value in RBX as 16 hexadecimal digits.
;
; Input : RBX = number to display
; ============================================================
DISPLAY_HEX:
    mov rsi, char_buff
    mov rcx, 16


DISPLAY_HEX_LOOP:
    rol rbx, 4

    mov dl, bl
    and dl, 0x0F

    cmp dl, 9
    jbe FORMAT_DEC_DIGIT

    add dl, 'A' - 10
    jmp STORE_HEX_DIGIT


FORMAT_DEC_DIGIT:
    add dl, '0'


STORE_HEX_DIGIT:
    mov byte [rsi], dl
    inc rsi
    dec rcx
    jnz DISPLAY_HEX_LOOP

    WRITE char_buff, 16
    ret


; ============================================================
; EXIT_PROGRAM
; Purpose: Terminate the program cleanly.
; ============================================================
EXIT_PROGRAM:
    mov rax, 60
    xor rdi, rdi
    syscall



section .note.GNU-stack noalloc noexec nowrite progbits

5. Multiplication using Successive Addition and Add-and-Shift

Pure assembly. Link with ld.

; ============================================================
; Program   : Multiplication using Successive Addition & Add-and-Shift
; Platform  : Linux x86-64
; Assembler : NASM
;
; Options:
;   1. Successive Addition (128-bit Hex result)
;   2. Add and Shift (128-bit Hex result)
;   3. Exit
; ============================================================



section .data
    ; Menu options and prompt strings
    menu        db 10, "----- MENU -----", 10
                db "1. Successive Addition", 10
                db "2. Add and Shift", 10
                db "3. Exit", 10
    menu_len    equ $ - menu

    ch_msg      db "Enter your choice : "
    ch_len      equ $ - ch_msg

    msg1        db 10, "Enter 1st 64-bit Hex Number: "
    len1        equ $ - msg1

    msg2        db "Enter 2nd 64-bit Hex Number : "
    len2        equ $ - msg2

    msg3        db "Multiplication Result (128-bit Hex) : "
    len3        equ $ - msg3

    newline     db 10



section .bss
    ; Buffers and 128-bit product registers
    char_buff   resb 17
    choice      resb 2

    num1        resq 1
    num2        resq 1
    res_high    resq 1
    res_low     resq 1



section .text
    global _start


; ============================================================
; MACRO: READ
; Purpose: Read characters from standard input.
;
; Parameters:
;   %1 = Address of receiving buffer
;   %2 = Maximum byte count to read
; ============================================================
%macro READ 2
    mov rax, 0
    mov rdi, 0
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro


; ============================================================
; MACRO: WRITE
; Purpose: Display a string to standard output.
;
; Parameters:
;   %1 = Address of string buffer
;   %2 = Length of string in bytes
; ============================================================
%macro WRITE 2
    mov rax, 1
    mov rdi, 1
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro


_start:
    ; --------------------------------------------------------
    ; Display menu and route user selection
    ; --------------------------------------------------------
PRINT_MENU:
    WRITE menu, menu_len
    WRITE ch_msg, ch_len
    READ choice, 2

    cmp byte [choice], '1'
    je SUCC_ADD_OP

    cmp byte [choice], '2'
    je ADD_SHIFT_OP

    cmp byte [choice], '3'
    je EXIT_PROGRAM

    jmp PRINT_MENU


; ============================================================
; OPTION 1: SUCCESSIVE ADDITION METHOD
; ============================================================
SUCC_ADD_OP:
    WRITE msg1, len1
    READ char_buff, 17
    dec rax                     ; Actual number of hex digits entered
    mov rcx, rax
    call ACCEPT_HEX
    mov [num1], rbx

    WRITE msg2, len2
    READ char_buff, 17
    dec rax
    mov rcx, rax
    call ACCEPT_HEX
    mov [num2], rbx

    mov rbx, [num1]             ; Multiplicand
    mov rcx, [num2]             ; Multiplier (counter)
    xor rax, rax                ; Lower 64 bits of result
    xor rdx, rdx                ; Upper 64 bits of result (carry)

    test rcx, rcx               ; If multiplier is 0, result is 0
    jz STORE_SUCC_RESULT


SUCC_ADD_LOOP:
    add rax, rbx
    jnc NO_CARRY_SUCC
    inc rdx                     ; Add carry into upper 64 bits


NO_CARRY_SUCC:
    dec rcx
    jnz SUCC_ADD_LOOP


STORE_SUCC_RESULT:
    mov [res_low], rax
    mov [res_high], rdx

    WRITE msg3, len3

    mov rbx, [res_high]
    call DISPLAY_HEX            ; Display upper 64 bits

    mov rbx, [res_low]
    call DISPLAY_HEX            ; Display lower 64 bits

    WRITE newline, 1
    jmp PRINT_MENU


; ============================================================
; OPTION 2: ADD AND SHIFT METHOD
; ============================================================
ADD_SHIFT_OP:
    WRITE msg1, len1
    READ char_buff, 17
    dec rax
    mov rcx, rax
    call ACCEPT_HEX
    mov [num1], rbx

    WRITE msg2, len2
    READ char_buff, 17
    dec rax
    mov rcx, rax
    call ACCEPT_HEX
    mov [num2], rbx

    mov rbx, [num1]             ; Multiplicand
    mov rsi, [num2]             ; Multiplier
    xor rax, rax                ; Lower 64 bits of accumulator
    xor rdx, rdx                ; Upper 64 bits of accumulator
    mov rcx, 64                 ; 64-bit counter


ADD_SHIFT_LOOP:
    shl rax, 1                  ; Shift 128-bit accumulator (RDX:RAX) left by 1
    rcl rdx, 1

    shl rsi, 1                  ; Shift MSB of multiplier into Carry Flag
    jnc NO_ADD_SHIFT

    add rax, rbx                ; If bit was 1, add multiplicand to accumulator
    jnc NO_ADD_SHIFT
    inc rdx                     ; Propagate carry to upper 64 bits


NO_ADD_SHIFT:
    dec rcx
    jnz ADD_SHIFT_LOOP

    mov [res_low], rax
    mov [res_high], rdx

    WRITE msg3, len3

    mov rbx, [res_high]
    call DISPLAY_HEX            ; Display upper 64 bits

    mov rbx, [res_low]
    call DISPLAY_HEX            ; Display lower 64 bits

    WRITE newline, 1
    jmp PRINT_MENU


; ============================================================
; PROCEDURE: ACCEPT_HEX
; Purpose: Converts an ASCII hexadecimal string into a 64-bit value.
;
; Input : char_buff, RCX = character count
; Output: RBX = 64-bit unsigned integer
; ============================================================
ACCEPT_HEX:
    mov rsi, char_buff
    xor rbx, rbx


ACCEPT_HEX_LOOP:
    xor rdx, rdx
    mov dl, byte [rsi]

    cmp dl, '9'
    jbe SUBTRACT_ZERO

    sub dl, 7


SUBTRACT_ZERO:
    sub dl, '0'
    shl rbx, 4
    add rbx, rdx

    inc rsi
    dec rcx
    jnz ACCEPT_HEX_LOOP

    ret


; ============================================================
; PROCEDURE: DISPLAY_HEX
; Purpose: Displays the 64-bit value in RBX as 16 hexadecimal digits.
;
; Input : RBX = number to display
; ============================================================
DISPLAY_HEX:
    mov rsi, char_buff
    mov rcx, 16


DISPLAY_HEX_LOOP:
    rol rbx, 4

    mov dl, bl
    and dl, 0x0F

    cmp dl, 9
    jbe FORMAT_DEC_DIGIT

    add dl, 'A' - 10
    jmp STORE_HEX_DIGIT


FORMAT_DEC_DIGIT:
    add dl, '0'


STORE_HEX_DIGIT:
    mov byte [rsi], dl
    inc rsi
    dec rcx
    jnz DISPLAY_HEX_LOOP

    WRITE char_buff, 16
    ret


; ============================================================
; EXIT_PROGRAM
; Purpose: Terminate the program cleanly.
; ============================================================
EXIT_PROGRAM:
    mov rax, 60
    xor rdi, rdi
    syscall



section .note.GNU-stack noalloc noexec nowrite progbits
nasm -f elf64 mult.asm
ld -o mult mult.o
./mult

6. Roots of a Quadratic Equation (All Possible Cases)

Uses C library (main, printf, scanf). Link with gcc -no-pie.

; ============================================================
; Program   : Roots of a Quadratic Equation using 80387 FPU
; Platform  : Linux x86-64
; Assembler : NASM
;
; Solves: a*x^2 + b*x + c = 0
; Uses C standard library functions: printf, scanf.
; ============================================================

extern printf, scanf



section .data
    ; Format specifiers and user prompts
    m1          db "%lf", 10, 0         ; Double-precision float format
    m2          db "%s", 10, 0          ; String format

    msg1        db "Enter the a,b,c", 0
    msg2        db 10, "Roots are", 0



section .bss
    ; Input coefficients (8-byte double precision)
    a           resb 8
    b           resb 8
    c           resb 8

    ; Temporary word and intermediate calculation variables
    temp        resw 1
    t1          resb 8
    t2          resb 8
    t3          resb 8
    t4          resb 8
    r1          resb 8
    r2          resb 8



section .text
    global main


; ============================================================
; MACRO: WRITE_STR
; Purpose: Display a string using printf.
;
; Parameters:
;   %1 = Format string address
;   %2 = Message string address
; ============================================================
%macro WRITE_STR 2
    push rbp
    xor rax, rax
    mov rdi, %1
    mov rsi, %2
    call printf
    pop rbp
%endmacro


; ============================================================
; MACRO: SCAN_FLOAT
; Purpose: Read a double-precision float using scanf.
;
; Parameters:
;   %1 = Format string address
;   %2 = Destination memory address
; ============================================================
%macro SCAN_FLOAT 2
    push rbp
    xor rax, rax
    mov rdi, %1
    mov rsi, %2
    call scanf
    pop rbp
%endmacro


; ============================================================
; MACRO: PRINT_FLOAT
; Purpose: Display a double-precision float using printf.
;
; Parameters:
;   %1 = Format string address
;   %2 = 64-bit floating-point memory location
; ============================================================
%macro PRINT_FLOAT 2
    push rbp
    mov rax, 1
    mov rdi, %1
    movsd xmm0, %2
    call printf
    pop rbp
%endmacro


main:
    ; --------------------------------------------------------
    ; 1. Read coefficients a, b, and c
    ; --------------------------------------------------------
    WRITE_STR m2, msg1
    SCAN_FLOAT m1, a
    SCAN_FLOAT m1, b
    SCAN_FLOAT m1, c


    ; --------------------------------------------------------
    ; 2. Initialize FPU and compute discriminant (b^2 - 4ac)
    ; --------------------------------------------------------
    finit

    ; t1 = b^2
    fld qword [b]
    fmul st0, st0
    fstp qword [t1]

    ; t2 = 4ac
    fld qword [a]
    fmul qword [c]
    mov word [temp], 4
    fimul word [temp]
    fstp qword [t2]

    ; t4 = b^2 - 4ac
    fld qword [t1]
    fsub qword [t2]
    fstp qword [t4]


    ; --------------------------------------------------------
    ; 3. Compute sqrt(|D|), -b, and 2a
    ; --------------------------------------------------------
    ; t1 = sqrt(|b^2 - 4ac|)
    fld qword [t4]
    fabs
    fsqrt
    fstp qword [t1]

    ; t2 = -b
    fld qword [b]
    fchs
    fstp qword [t2]

    ; t3 = 2a
    fld qword [a]
    mov word [temp], 2
    fimul word [temp]
    fstp qword [t3]


    ; --------------------------------------------------------
    ; 4. Calculate and print both roots
    ; --------------------------------------------------------
    WRITE_STR m2, msg2

    ; Root 1 = (-b + sqrt(|D|)) / 2a
    fld qword [t2]
    fadd qword [t1]
    fdiv qword [t3]
    fstp qword [r1]
    PRINT_FLOAT m1, [r1]


EQUAL_ROOT:
    ; Root 2 = (-b - sqrt(|D|)) / 2a
    fld qword [t2]
    fsub qword [t1]
    fdiv qword [t3]
    fstp qword [r2]
    PRINT_FLOAT m1, [r2]

    xor rax, rax
    ret



section .note.GNU-stack noalloc noexec nowrite progbits
nasm -f elf64 quad.asm
gcc -no-pie -o quad quad.o
./quad
